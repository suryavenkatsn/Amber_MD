export CUDA_VISIBLE_DEVICES=0         # [0,1,2,3]

# Ligand Parameterisation
antechamber -i BWC1063_1.mol2 -fi mol2 -o BWC1063_1-ac.mol2 -fo mol2 -nc 1 -c bcc -dr no

$AMBERHOME/bin/./parmchk2 -i BWC1063_1-ac.mol2 -f mol2 -o frcmod.UNK

# Ligand Preparation
$AMBERHOME/bin/./tleap -f /opt/amber18/dat/leap/cmd/oldff/leaprc.ff99SB
source /opt/amber18/dat/leap/cmd/leaprc.zinc
source /opt/amber18/dat/leap/cmd/leaprc.gaff
UNK = loadmol2 BWC1063_1-ac.mol2
check UNK
loadamberparams frcmod.UNK
saveoff UNK UNK.lib
saveamberparm UNK UNK.prmtop UNK.inpcrd
savepdb UNK UNKlig.pdb
quit

# Protein preparation
$AMBERHOME/bin/./tleap -f /opt/amber18/dat/leap/cmd/oldff/leaprc.ff99SB
source /opt/amber18/dat/leap/cmd/leaprc.zinc
source /opt/amber18/dat/leap/cmd/leaprc.gaff
pro = loadpdb 7bv2_prep_5.pdb
saveamberparm pro rec.prmtop rec.inpcrd
savepdb  pro rec.pdb
quit

# Make complex by combining rec and UNK
sh ./make_complex.sh


# Complex Preparation
$AMBERHOME/bin/./tleap -f /opt/amber18/dat/leap/cmd/oldff/leaprc.ff99SB
source /opt/amber18/dat/leap/cmd/leaprc.zinc
source /opt/amber18/dat/leap/cmd/leaprc.gaff
loadamberparams frcmod.UNK
loadoff UNK.lib
complex = loadpdb complex.pdb
saveamberparm  complex com.prmtop com.inpcrd
savepdb complex com.pdb
quit

# Ligand and Complex Minimization
sh ./int.sh &

# Converting rst to pdb
$AMBERHOME/bin/./ambpdb -p com.prmtop -c complexmin.rst > complexmin.pdb

# Solvation and Adding Ions
$AMBERHOME/bin/./tleap -f /opt/amber18/dat/leap/cmd/oldff/leaprc.ff99SB
source /opt/amber18/dat/leap/cmd/leaprc.zinc
source /opt/amber18/dat/leap/cmd/leaprc.gaff
loadamberparams frcmod.UNK
loadoff UNK.lib
complex = loadpdb complexmin.pdb
solvatebox complex TIP3PBOX 10.0
charge complex
addIons complex Na+ 0
saveamberparm complex com-sol1.prmtop com-sol1.inpcrd
savepdb complex com-sol1.pdb
quit

# Final simulation
sh ./fin.sh &

# nc to mdcrd
cpptraj com-sol1.prmtop <<EOF
trajin prod1.nc 0 10000000 1
reference com-sol1.inpcrd
trajout prod1.mdcrd
EOF
